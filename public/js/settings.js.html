document.addEventListener('DOMContentLoaded', function () { // TODO Call srver-side function to load the settings // @ts-ignore google.script.run.withSuccessHandler(loadSettings).getSettings(); }); // Helper function to create a list item function createListItem(content, listType) { const listItem = document.createElement('li'); listItem.className = 'list-group-item d-flex justify-content-between align-items-xl-center'; // Create the span element const span = document.createElement('span'); span.textContent = content; // Create the button element const button = document.createElement('button'); button.className = 'btn btn-danger'; button.type = 'button'; button.onclick = deleteListItem; // Create the icon element const icon = document.createElement('i'); icon.className = 'fa fa-close'; // Append the icon to the button button.appendChild(icon); // Append the image for class list items if (listType === 'class-list') { const img = document.createElement('img'); img.src = 'https://upload.wikimedia.org/wikipedia/commons/1/19/Google_Classroom_Logo.svg'; img.width = 50; img.height = 50; listItem.appendChild(img); } // Append the span and button to the list item listItem.appendChild(span); listItem.appendChild(button); return listItem; } // Updated loadSettings function function loadSettings(settings) { // Set the radio button for shift if (settings._shift) document.getElementById('shiftCheck-' + settings._shift).checked = true; // Set the time count input value if (settings._max_classes && settings._max_classes > 0) document.getElementById('time-count').value = settings._max_classes; // Set the declarations list const declList = document.getElementById("declarations-list"); declList.innerHTML = ""; (settings._declarations_templates || []).forEach(decl => { const div = document.createElement("div"); div.className = "input-group mb-2"; div.dataset.url = decl.url; div.dataset.name = decl.name; div.innerHTML = ` ${decl.name} `; declList.appendChild(div); }); // Set the classes list if (settings._classes && settings._classes.length > 0) { const classesList = document.getElementById('class-list'); classesList.innerHTML = ''; // Clear the list before adding new items settings._classes.forEach(className => { const listItem = createListItem(className, 'class-list'); classesList.appendChild(listItem); }); } // Set the rooms list if (settings._rooms && settings._rooms.length > 0) { const roomsList = document.getElementById('rooms-list'); roomsList.innerHTML = ''; // Clear the list before adding new items settings._rooms.forEach(roomName => { const listItem = createListItem(roomName, 'rooms-list'); roomsList.appendChild(listItem); }); } } // Updated addListItem function function addListItem(input) { const listType = input.name === 'class-name' ? 'class-list' : input.name === 'room-name' ? 'rooms-list' : 'declarations-list'; const listItem = createListItem(input.value, listType); // Append the list item to the desired list document.getElementById(listType).appendChild(listItem); input.value = ''; } function deleteListItem(element) { // Remove the list item from the list this.parentNode.remove(); } function extractFileIdFromUrl(url) { const match = url.match(/[-\w]{25,}/); return match ? match[0] : null; } function addDeclarationTemplate() { const input = document.getElementById("declaration-input"); const url = input.value.trim(); if (!url) return; const fileId = extractFileIdFromUrl(url); if (!fileId) { alert("Невалиден Google Docs линк"); return; } google.script.run.withSuccessHandler(name => { const list = document.getElementById("declarations-list"); const div = document.createElement("div"); div.className = "input-group mb-2"; div.dataset.url = url; div.dataset.name = name; div.innerHTML = ` ${name} `; list.appendChild(div); input.value = ""; }).getGoogleDocName(fileId); } function removeDeclarationTemplate(button) { const div = button.parentElement; div.remove(); } function getDeclarationTemplates() { const list = document.getElementById("declarations-list").children; const result = []; for (const item of list) { result.push({ url: item.dataset.url, name: item.dataset.name }); } return result; } function saveSettings() { const shift = document.querySelector('input[name="shift"]:checked').value; const timeCount = document.getElementById('time-count').value; const declarations = getDeclarationTemplates(); const classes = Array.from(document.querySelectorAll('#class-list li span')).map(span => span.textContent); const rooms = Array.from(document.querySelectorAll('#rooms-list li span')).map(span => span.textContent); google.script.run.saveSettings(shift, timeCount, declarations, classes, rooms); } document.getElementById('shiftCheck-first').addEventListener('change', function () { if (this.checked) { saveSettings(); } }); document.getElementById('shiftCheck-second').addEventListener('change', function () { if (this.checked) { saveSettings(); } }); document.getElementById('time-count').addEventListener('change', function () { saveSettings(); }); // Select all relevant lists to observe const lists = document.querySelectorAll("#class-list, #declarations-list, #rooms-list"); // Create a MutationObserver instance const observer = new MutationObserver((mutations) => { mutations.forEach((mutation) => { console.log("Something changed!", mutation); saveSettings(); // Call saveSettings when any list changes }); }); // Start observing each list for changes lists.forEach((list) => { observer.observe(list, { childList: true, subtree: false }); });
