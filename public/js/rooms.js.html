/** * Зарежда графика на стаите. * Извиква сървърната функция `getRoomUsageGrid` и предава резултата на `renderRoomSchedule`. */ function loadRoomSchedule() { google.script.run.withSuccessHandler(renderRoomSchedule).getRoomUsageGrid(); // Извиква сървърната функция и обработва резултата. } /** * Рендира графика на стаите в HTML. * Създава табове за дните от седмицата и таблици за всяка смяна и стая. * @param data - Обект с данни за графика на стаите и настройките. */ function renderRoomSchedule(data) { const { grid, settings } = data; // Извлича графика и настройките от данните. const container = document.getElementById("room-schedule-container"); // Контейнер за графика на стаите. container.innerHTML = ""; // Изчиства съдържанието на контейнера. const days = ["monday", "tuesday", "wednesday", "thursday", "friday"]; // Дни от седмицата. const dayNames = { // Преводи на дните от седмицата. monday: "Понеделник", tuesday: "Вторник", wednesday: "Сряда", thursday: "Четвъртък", friday: "Петък" }; // Създава навигационни табове за дните. const tabsNav = document.createElement("ul"); tabsNav.className = "nav nav-tabs mb-3"; // Класове за стилизиране на табовете. tabsNav.id = "room-day-tabs"; tabsNav.role = "tablist"; // Създава съдържание за табовете. const tabsContent = document.createElement("div"); tabsContent.className = "tab-content"; days.forEach((day, i) => { // Създава бутон за таба на текущия ден. const tabId = `room-tab-${day}`; const tabButton = document.createElement("li"); tabButton.className = "nav-item"; tabButton.innerHTML = ` `; tabsNav.appendChild(tabButton); // Добавя бутона към навигацията. // Създава съдържание за таба на текущия ден. const tabPane = document.createElement("div"); tabPane.className = `tab-pane fade ${i === 0 ? "show active" : ""}`; // Активира първия таб. tabPane.id = tabId; tabPane.role = "tabpanel"; // Определя смените (първа и/или втора). const shifts = settings._shift === "second" ? ["first", "second"] : ["first"]; shifts.forEach(shift => { // Създава заглавие за смяната. const shiftLabel = document.createElement("h6"); shiftLabel.className = "mt-3"; shiftLabel.textContent = shift === "first" ? "Първа смяна" : "Втора смяна"; tabPane.appendChild(shiftLabel); // Създава таблица за графика на стаите. const table = document.createElement("table"); table.className = "table table-bordered bg-white"; // Създава заглавията на таблицата. const thead = document.createElement("thead"); thead.innerHTML = "СтаяЧасКлас"; table.appendChild(thead); const tbody = document.createElement("tbody"); const shiftData = grid[day][shift]; // Взима данните за текущата смяна. // Обхожда всяка стая и добавя редове за часовете. Object.keys(shiftData).forEach(room => { const entries = shiftData[room]; const sorted = [...entries].sort((a, b) => Number(a.time) - Number(b.time)); // Сортира записите по час. if (sorted.length === 0) return; // Пропуска стаи без записи. sorted.forEach((entry, index) => { const row = document.createElement("tr"); if (index === 0) { // Създава клетка за името на стаята (обединена за всички часове). const roomCell = document.createElement("td"); roomCell.textContent = room; roomCell.rowSpan = sorted.length; // Обединява клетките за стаята. row.appendChild(roomCell); } // Създава клетка за часа. const timeCell = document.createElement("td"); timeCell.textContent = entry.time; row.appendChild(timeCell); // Създава клетка за класа. const groupCell = document.createElement("td"); groupCell.textContent = entry.group || ""; row.appendChild(groupCell); tbody.appendChild(row); // Добавя реда в таблицата. }); }); table.appendChild(tbody); // Добавя тялото на таблицата. tabPane.appendChild(table); // Добавя таблицата в таба. }); tabsContent.appendChild(tabPane); // Добавя съдържанието на таба. }); container.appendChild(tabsNav); // Добавя навигацията към контейнера. container.appendChild(tabsContent); // Добавя съдържанието на табовете към контейнера. }
