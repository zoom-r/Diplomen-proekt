// Зареждане на всички училища при стартиране function loadWorkspaces() { google.script.run.withSuccessHandler(renderWorkspaces).getAllWorkspaces(); } // Рендиране на училищата function renderWorkspaces(workspaces) { const list = document.getElementById('workspacesList'); list.innerHTML = ''; workspaces.forEach(ws => { const tr = document.createElement('tr'); tr.innerHTML = ` ${ws.school} `; list.appendChild(tr); }); } // Създаване на ново училище function createWorkspace() { const name = document.getElementById('schoolNameInput').value.trim(); if (!name) { alert('Моля, въведете име на училище.'); return; } google.script.run.withSuccessHandler(() => { document.getElementById('schoolNameInput').value = ''; loadWorkspaces(); }).createWorkspace(name); } // Отваряне на модала за добавяне на администратор function openAddAdminModal(workspaceId) { document.getElementById('selectedWorkspaceId').value = workspaceId; document.getElementById('adminNamesInput').value = ''; document.getElementById('adminEmailInput').value = ''; document.getElementById('adminPhoneInput').value = ''; document.getElementById('adminPositionInput').value = ''; const modal = new bootstrap.Modal(document.getElementById('addAdminModal')); modal.show(); } // Създаване на нов администратор function createAdmin() { const data = { workspaceId: document.getElementById('selectedWorkspaceId').value, names: document.getElementById('adminNamesInput').value.trim(), email: document.getElementById('adminEmailInput').value.trim(), phone: document.getElementById('adminPhoneInput').value.trim(), position: document.getElementById('adminPositionInput').value.trim() }; if (!data.names || !data.email || !data.phone || !data.position) { alert('Моля, попълнете всички полета.'); return; } google.script.run.withSuccessHandler(() => { const modal = bootstrap.Modal.getInstance(document.getElementById('addAdminModal')); modal.hide(); loadWorkspaces(); }).createAdminUser(data); } function deleteWorkspace(workspaceId) { if (!confirm('Сигурни ли сте, че искате да изтриете това училище?')) return; google.script.run.withSuccessHandler(() => { loadWorkspaces(); }).deleteWorkspace(workspaceId); } // Автоматично зареждане при отваряне на страницата document.addEventListener('DOMContentLoaded', loadWorkspaces);
